CREATE DATABASE SpaceSurvivor;
USE SpaceSurvivor;
CREATE TABLE Players (
    player_id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    fecha_registro DATETIME NOT NULL
);


CREATE TABLE Sessions (
    session_id INT AUTO_INCREMENT PRIMARY KEY,
    player_id INT NOT NULL,
    fecha_inicio DATETIME NOT NULL,
    fecha_fin DATETIME,
    dificultad VARCHAR(20) NOT NULL,
    FOREIGN KEY (player_id) REFERENCES Players(player_id)
);


CREATE TABLE EnemyTypes (
    enemy_type_id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    vida_base INT NOT NULL CHECK (vida_base > 0),
    dano INT NOT NULL CHECK (dano >= 0),
    velocidad DECIMAL(6,2) NOT NULL CHECK (velocidad > 0)
);


CREATE TABLE EnemyInstances (
    instance_id INT AUTO_INCREMENT PRIMARY KEY,
    session_id INT NOT NULL,
    enemy_type_id INT NOT NULL,
    x DECIMAL(8,2) NOT NULL,
    y DECIMAL(8,2) NOT NULL,
    vida_actual INT NOT NULL CHECK (vida_actual >= 0),
    FOREIGN KEY (session_id) REFERENCES Sessions(session_id),
    FOREIGN KEY (enemy_type_id) REFERENCES EnemyTypes(enemy_type_id)
);


CREATE TABLE Weapons (
    weapon_id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    dano INT NOT NULL CHECK (dano >= 0),
    cadencia DECIMAL(6,2) NOT NULL CHECK (cadencia > 0),
    municion_max INT NOT NULL CHECK (municion_max > 0)
);


CREATE TABLE CombatLogs (
    log_id INT AUTO_INCREMENT PRIMARY KEY,
    session_id INT NOT NULL,
    timestamp DATETIME NOT NULL,
    tipo_evento VARCHAR(30) NOT NULL,
    entidad_origen VARCHAR(50),
    entidad_destino VARCHAR(50),
    valor INT,
    FOREIGN KEY (session_id) REFERENCES Sessions(session_id)
);


CREATE TABLE AmmoInventory (
    inventory_id INT AUTO_INCREMENT PRIMARY KEY,
    session_id INT NOT NULL,
    weapon_id INT NOT NULL,
    cantidad_actual INT NOT NULL CHECK (cantidad_actual >= 0),
    FOREIGN KEY (session_id) REFERENCES Sessions(session_id),
    FOREIGN KEY (weapon_id) REFERENCES Weapons(weapon_id)
);


CREATE TABLE Leaderboard (
    leaderboard_id INT AUTO_INCREMENT PRIMARY KEY,
    session_id INT NOT NULL,
    puntaje_final INT NOT NULL CHECK (puntaje_final >= 0),
    tiempo_supervivencia INT NOT NULL CHECK (tiempo_supervivencia >= 0),
    enemigos_eliminados INT NOT NULL CHECK (enemigos_eliminados >= 0),
    FOREIGN KEY (session_id) REFERENCES Sessions(session_id)
);
